  <script>
    window.onload = function() {
      loadAllData();
    };

    function loadAllData() {
      google.script.run.withSuccessHandler(populateDates).getAvailableDates();
      google.script.run.withSuccessHandler(renderViewTable).getViewData();
    }

    function populateDates(dates) {
      var select = document.getElementById('dateSelect');
      var btn = document.getElementById('submitBtn');
      select.innerHTML = '';
      
      if (!dates || dates.length === 0) {
        select.innerHTML = '<option value="">--- วันลาถูกจองเต็มทุกวันแล้ว ---</option>';
        select.disabled = true;
        btn.disabled = true;
      } else {
        select.innerHTML = '<option value="">-- กรุณาเลือกวันที่ต้องการลา --</option>';
        dates.forEach(function(d) {
          var opt = document.createElement('option');
          opt.value = d;
          opt.textContent = d;
          select.appendChild(opt);
        });
        select.disabled = false;
        btn.disabled = false;
      }
    }

    function renderViewTable(rows) {
      var tbody = document.getElementById('viewTableBody');
      tbody.innerHTML = '';

      if (!rows || rows.length === 0) {
        tbody.innerHTML = '<tr><td colspan="5" class="py-4 text-muted">ไม่พบข้อมูลในตาราง</td></tr>';
        return;
      }

      rows.forEach(function(r) {
        var shift = (r[0] || "").trim();
        var dayName = (r[1] || "").trim();
        var dateVal = (r[2] || "").trim();
        var leavePerson = (r[3] || "").trim();
        var backupPerson = (r[4] || "").trim();

        var isHoliday = (shift === 'วันหยุด' || leavePerson === 'วันหยุด');
        var isTaken = (leavePerson !== 'ว่าง' && leavePerson !== 'วันหยุด' && leavePerson !== '');

        var shiftBadge = '<span class="badge-shift shift-n1">' + shift + '</span>';
        if (shift === 'วันหยุด') {
          shiftBadge = '<span class="badge-shift shift-holiday">วันหยุด</span>';
        } else if (shift === 'N2') {
          shiftBadge = '<span class="badge-shift shift-n2">N2</span>';
        } else if (shift === 'N3') {
          shiftBadge = '<span class="badge-shift shift-n3">N3</span>';
        }

        var leaveDisplay = '-';
        if (leavePerson === 'ว่าง') {
          leaveDisplay = '<span class="status-empty">🟢 ว่าง</span>';
        } else if (leavePerson === 'วันหยุด') {
          leaveDisplay = '<span class="status-holiday-text"> วันหยุด</span>';
        } else if (leavePerson !== '') {
          leaveDisplay = '<span class="status-taken">👤 ' + leavePerson + '</span>';
        }

        var textClass = '';
        if (isHoliday) {
          textClass = ' class="text-holiday-danger"';
        } else if (isTaken) {
          textClass = ' class="text-taken-blue"';
        }

        var tr = document.createElement('tr');
        if (isHoliday) {
          tr.style.backgroundColor = '#fef2f2';
        } else if (isTaken) {
          tr.style.backgroundColor = '#eff6ff';
        }

        tr.innerHTML = 
          '<td>' + shiftBadge + '</td>' +
          '<td' + textClass + '>' + dayName + '</td>' +
          '<td' + textClass + '>' + dateVal + '</td>' +
          '<td>' + leaveDisplay + '</td>' +
          '<td>' + (backupPerson ? '<span class="' + (isTaken ? 'text-taken-blue' : 'text-secondary') + ' fw-semibold">' + backupPerson + '</span>' : '<span class="text-muted">-</span>') + '</td>';

        tbody.appendChild(tr);
      });
    }

    function handleFormSubmit(e) {
      e.preventDefault();
      var name = document.getElementById('userName').value.trim();
      var date = document.getElementById('dateSelect').value;
      var phone = document.getElementById('userPhone').value.trim();
      var backup = document.getElementById('backupName').value.trim();
      var btn = document.getElementById('submitBtn');
      var statusDiv = document.getElementById('statusMessage');

      btn.disabled = true;
      btn.textContent = '⏳ กำลังบันทึกและอัปเดตตาราง...';
      statusDiv.innerHTML = '';

      google.script.run
        .withSuccessHandler(function(response) {
          populateDates(response.updatedDates);
          renderViewTable(response.updatedView);
          btn.textContent = 'ยืนยันการจองวันลา';
          
          if (response.success) {
            showStatus(response.message, 'success');
            document.getElementById('userName').value = '';
            document.getElementById('userPhone').value = '';
            document.getElementById('backupName').value = '';
          } else {
            showStatus(response.message, 'danger');
          }
        })
        .withFailureHandler(function(error) {
          showStatus('เกิดความผิดพลาด: ' + error.message, 'danger');
          btn.disabled = false;
          btn.textContent = 'ยืนยันการจองวันลา';
        })
        .submitBooking(name, date, backup, phone);
    }

    function showStatus(msg, type) {
      var statusDiv = document.getElementById('statusMessage');
      statusDiv.innerHTML = '<div class="alert alert-' + type + ' m-0">' + msg + '</div>';
    }
